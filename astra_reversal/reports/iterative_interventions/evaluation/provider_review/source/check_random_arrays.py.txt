import numpy as np
from astra_reversal.interventions import random_noise_proposal
from astra_reversal.records import digest
episode='libero_spatial_ood:seed19:task2:state0'
rng=np.random.default_rng(np.random.SeedSequence([19,int(digest(episode)[:8],16),3]))
for iteration in range(2,5):
    local=random_noise_proposal(np.zeros((8,1,10,32),dtype=np.float32),rng)
REMOTE=REMOTE.replace('__LOCAL_REGENERATED__',repr({'coefficients':local['coefficients'],'perturbation_scale':local['perturbation_scale']})).replace('__LOCAL_NUMPY_VERSION__',repr(np.__version__))
directory = Path(__file__).resolve().parent
command = 'cd /workspace/astra-reversal\nsource /workspace/astra-reversal/emimic/bin/activate\n' + shlex.join(['python','-c',REMOTE])
master, slave = pty.openpty()
fcntl.ioctl(slave,termios.TIOCSWINSZ,struct.pack('HHHH',40,240,0,0))
process = subprocess.Popen(['osmo','workflow','exec','astra-pi05-interventions-eval-20260924-1','worker6','--connect-timeout','20','--entry',shlex.join(['/bin/bash','-lc',command])],stdin=slave,stdout=slave,stderr=slave,start_new_session=True)
os.close(slave)
received=bytearray()
deadline=time.monotonic()+80
try:
    while time.monotonic()<deadline:
        ready,_,_=select.select([master],[],[],1)
        if ready:
            try: block=os.read(master,65536)
            except OSError as exc:
                if exc.errno==errno.EIO: break
                raise
            if not block: break
            received.extend(block)
        elif process.poll() is not None: break
    if process.poll() is None: process.terminate()
    process.wait(timeout=10)
finally:
    os.close(master)
    if process.poll() is None: process.kill()
log=directory/'random-array-check-console.log'
log.write_bytes(received);log.chmod(0o600)
records=[json.loads(line.split('ASTRA_ARRAY_CHECK:',1)[1]) for line in received.decode(errors='replace').splitlines() if line.startswith('ASTRA_ARRAY_CHECK:')]
assert process.returncode==0 and len(records)==1, f'Read-only check not completed; local exit={process.returncode}; inspect private console'
result=records[0]
result['auditor_sha256']=hashlib.sha256(Path(__file__).read_bytes()).hexdigest()
output=directory/'random-coefficient-array-review.json'
output.write_text(json.dumps(result,indent=2,allow_nan=False)+'\n');output.chmod(0o600)
print(json.dumps({'path':str(output),'sha256':hashlib.sha256(output.read_bytes()).hexdigest(),'status':result['status'],'array_files':result['array_files_verified'],'generations':result['generation_count']}))
